def LegendreK6.CSv2 : Prop :=
∀ (i : Fin 12) (σ T : ℝ),
  (16 + ↑↑i) / 32 ≤ σ →
    σ ≤ (17 + ↑↑i) / 32 →
      3000000000000 ≤ T →
        ↑(LegendreK6.densityZeroCount σ T) ≤
          LegendreK6.csB1 i * T ^ LegendreK6.csExponent σ * Real.log T ^ LegendreK6.csLogExponent σ +
              LegendreK6.csB2 i * Real.log T ^ 2 +
            LegendreK6.csB3 i * Real.log T
def LegendreK6.RosserZeroCount : Prop :=
∀ (T : ℝ),
  2 ≤ T →
    |↑(LegendreK6.rosserCount T) - (T / (2 * Real.pi) * Real.log (T / (2 * Real.pi)) - T / (2 * Real.pi) + 7 / 8)| ≤
      137 / 1000 * Real.log T + 443 / 1000 * Real.log (Real.log T) + 397 / 250
def LegendreK6.CHJIIShortInterval : Prop :=
∀ (x h T : ℝ),
  0 < x →
    40 ≤ Real.log x →
      0 ≤ h →
        max 51 (Real.log x ^ 2) < T →
          T < (√x - 2) / 4 →
            ∃ Tstar,
              T ≤ Tstar ∧
                Tstar ≤ 2 * T ∧
                  ‖↑(Chebyshev.psi (x + h) - Chebyshev.psi x - h) + LegendreK6.truncatedZeroIncrement x (x + h) Tstar‖ ≤
                    1447 / 500 / T * ((x + h) * Real.log (x + h) + x * Real.log x)
